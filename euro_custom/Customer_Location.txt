// ==========================================================
// القسم 1: ربط الأحداث (Event Triggers) في شاشة العميل Customer
// تم تنظيف التكرار وربط كل زرار بوظيفته مرة واحدة فقط
// ==========================================================

frappe.ui.form.on('Customer', {
    refresh: function(frm) {
        
        // 1. ربط زرار [تحديث الخريطة الداخلية] - كان متكرراً وتم دمجه
        if (frm.fields_dict['open_in_google_maps']) {
            frm.fields_dict['open_in_google_maps'].$wrapper.find('button').off('click').on('click', function() {
                execute_smart_map_sync(frm);
            });
        }

        // 2. ربط زرار [فتح جوجل ماب في صفحة خارجية]
        if (frm.fields_dict['view_external_map']) {
            frm.fields_dict['view_external_map'].$wrapper.find('button').off('click').on('click', function() {
                open_location_externally(frm);
            });
        }
    }
});


// ==========================================================
// القسم 2: الدالة الرئيسية لتنفيذ مزامنة الموقع (Smart Map Sync)
// (نفس منطق الكود الخاص بك دون أي تعديل في الوظائف)
// ==========================================================

function execute_smart_map_sync(frm) {
    if (!frm.doc.whatsapp_location_url) {
        frappe.msgprint(__("من فضلك ضع الرابط أولاً"));
        return;
    }

    let url = frm.doc.whatsapp_location_url;

    // استخراج الإحداثيات
    let match = url.match(/([-+]?\d+\.\d+)\s*,\s*([-+]?\d+\.\d+)/) || 
                url.match(/q=([-+]?\d+\.\d+),([-+]?\d+\.\d+)/) || 
                url.match(/@([-+]?\d+\.\d+),([-+]?\d+\.\d+)/);

    if (match) {
        let lat = parseFloat(match[1]);
        let lng = parseFloat(match[2]);

        let location_json = {
            "type": "FeatureCollection",
            "features": [{
                "type": "Feature",
                "properties": {},
                "geometry": { "type": "Point", "coordinates": [lng, lat] }
            }]
        };
        frm.set_value('customer_map', JSON.stringify(location_json));

        setTimeout(() => {
            let map_field = frm.fields_dict.customer_map;
            if (map_field && map_field.map) {
                let map = map_field.map;
                if (map._layers) {
                    Object.keys(map._layers).forEach(function(layerId) {
                        let layer = map._layers[layerId];
                        if (layer instanceof L.Marker) {
                            map.removeLayer(layer);
                        }
                    });
                }
                L.marker([lat, lng]).addTo(map);
                map.invalidateSize();
                map.setView([lat, lng], 18, { animate: true });
            }
        }, 300);

        fetch(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${lat}&lon=${lng}&accept-language=ar`)
            .then(res => res.json())
            .then(data => { 
                if (data && data.display_name) {
                    frm.set_value('address', data.display_name); 
                }
            });

        frappe.show_alert({message: __('تم تحديث الموقع وتثبيت الخريطة'), indicator: 'green'});
    } else {
        frappe.msgprint(__("الرابط لا يحتوي على إحداثيات صحيحة"));
    }
}


// ==========================================================
// القسم 3: وظيفة الزرار الخاص بالفتح الخارجي
// ==========================================================

function open_location_externally(frm) {
    if (frm.doc.whatsapp_location_url) {
        window.open(frm.doc.whatsapp_location_url, '_blank');
    } else {
        frappe.msgprint(__("الحقل فارغ، لا يوجد رابط لفتحه"));
    }
}
